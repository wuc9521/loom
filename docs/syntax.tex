\section*{Formal Syntax of Logic Learning Language}

\begin{align*}
     \Program             ::=~       & \Stmt^+                                                                                \\
     \\
     \Stmt                ::=~       & \SetLogic{\Logic}                                                         \\
     \mid~                           & \DeclareSort{\Sort}{\Arity}                                                \\
     \mid~                           & \Constraint{\Bexp}                                                                      \\
     \mid~                           & \CheckSynth                                                                              \\
     \\
     \SigDef           ::=~          & \SigStmt^+                                                                             \\
     \SigStmt       ::=~             & \DeclareFun{\Func}{\Sort^*}{\Sort}                            \\
     \\
     \Sort                ::=~       & \Ident                                                                                 \\
     \Func           ::=~            & \Ident                                                                                 \\
     \\
     \FuncInterp ::=~                & \Func~\InterpValue                                                                     \\
     \InterpStmt  ::=~               & \Lparen\DefineUniverse~\UniverseDef\Rparen                                             \\
     \mid~                           & \Lparen\Interpret~\Lparen\FuncInterp^*\Rparen\Rparen                                   \\
     \InterpValue ::=~               & \Ident \mid \Match                                                                     \\
     \UniverseDef         ::=~       & \Lparen\Ident~\{ \Ident^+ \}\Rparen                                                    \\
     \\
     \Match               ::=~       & \Lparen\texttt{match}~\MatchCase^+\Rparen                                              \\
     \MatchCase           ::=~       & \Lparen\Lparen\InputValue^+\Rparen~\OutValue\Rparen                                    \\                                                      \\
     \\
     \Grammar             ::=~       & \GrammarRule^+                                                                         \\
     \GrammarRule         ::=~       & \Lparen\NonTermName~\Sort\Rparen                                                             \\
     \mid~                           & \Lparen\NonTermName~\ProductionRules\Rparen                                                  \\
     \ProductionRules     ::=~       & \Ident^+                                                                               \\
     \\
     \Bexp                 ::=~      & \Lparen\And~\Bexp~\Bexp\Rparen                                                         \\
     \mid~                           & \Lparen\Or~\Bexp~\Bexp\Rparen                                                          \\
     \mid~                           & \Lparen\Not~\Bexp\Rparen                                                               \\
     \mid~                           & \Lparen\texttt{models}~\Model~\Form\Rparen                                             \\
     \mid~                           & \Bool
     \\
     \InputValue                ::=~ & \Ident \mid~\texttt{\_}                                                                \\
     \OutValue ::=~                  & \Ident \mid~\Bool                                                                      \\
     \Logic               ::=~       & \Ident                                                                                 \\
     \Nat                 ::=~       & \texttt{0} \mid \texttt{1} \mid \dots                                                  \\
     \Bool                ::=~       & \True~\mid~\False                                                                      \\
     \Ident               ::=~       & \texttt{a} \mid \texttt{b} \mid \dots~\Lparen\text{any valid identifier}\Rparen        \\
     \Arity               ::=~       & \Nat                                                                                   \\
     \NonTermName         ::=~       & \Ident                                                                                 \\
\end{align*}
